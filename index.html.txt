    <script>
        // --- State Management ---
        const STORAGE_KEY = 'baby_tracker_logs';
        let logs = JSON.parse(localStorage.getItem(STORAGE_KEY)) || [];
        let currentTab = 'day';
        let trendsChart = null;
        let editingLogId = null;

        // --- Core Logic ---
        function saveLogs() {
            localStorage.setItem(STORAGE_KEY, JSON.stringify(logs));
            updateDashboard();
            renderHistory();
            updateChart();
        }

        function getLocalIsoString() {
            const now = new Date();
            now.setMinutes(now.getMinutes() - now.getTimezoneOffset());
            return now.toISOString().slice(0, 16);
        }

        function logPoop() {
            const dtInput = document.getElementById('poop-datetime').value;
            const timestamp = dtInput ? new Date(dtInput).getTime() : Date.now();

            logs.unshift({
                id: Date.now().toString(),
                type: 'poop',
                timestamp: timestamp
            });
            
            // Re-sort in case of backdated entry
            logs.sort((a, b) => b.timestamp - a.timestamp);
            
            saveLogs();
            closePoopModal();
            showToast('💩 Poop logged!');
        }

        function logFeeding() {
            const volInput = document.getElementById('volume-input');
            const volume = parseInt(volInput.value);
            const dtInput = document.getElementById('feed-datetime').value;
            const timestamp = dtInput ? new Date(dtInput).getTime() : Date.now();
            
            if (isNaN(volume) || volume <= 0) {
                // Show simple feedback instead of alert
                volInput.classList.add('border-red-400');
                setTimeout(() => volInput.classList.remove('border-red-400'), 1000);
                return;
            }

            logs.unshift({
                id: Date.now().toString(),
                type: 'feed',
                volume: volume,
                timestamp: timestamp
            });
            
            // Re-sort in case of backdated entry
            logs.sort((a, b) => b.timestamp - a.timestamp);
            
            saveLogs();
            closeFeedModal();
            showToast(`🍼 ${volume}ml logged!`);
        }

        function clearHistory() {
            // Custom confirmation dialog instead of window.confirm
            if(logs.length === 0) return;
            
            // Simple inline confirmation for this implementation
            const btn = document.querySelector('button[onclick="clearHistory()"]');
            if (btn.innerText === "Sure?") {
                logs = [];
                saveLogs();
                btn.innerText = "Clear All";
                btn.classList.remove('bg-red-500', 'text-white');
                btn.classList.add('bg-red-50', 'text-red-400');
            } else {
                btn.innerText = "Sure?";
                btn.classList.remove('bg-red-50', 'text-red-400');
                btn.classList.add('bg-red-500', 'text-white');
                setTimeout(() => {
                    btn.innerText = "Clear All";
                    btn.classList.remove('bg-red-500', 'text-white');
                    btn.classList.add('bg-red-50', 'text-red-400');
                }, 3000);
            }
        }

        function exportData() {
            if (logs.length === 0) {
                showToast("No data to export!");
                return;
            }

            // Create CSV header
            let csvContent = "data:text/csv;charset=utf-8,";
            csvContent += "Date,Time,Type,Volume (ml)\n";

            // Add rows
            logs.forEach(log => {
                const dateObj = new Date(log.timestamp);
                const dateStr = dateObj.toLocaleDateString();
                const timeStr = dateObj.toLocaleTimeString();
                const typeStr = log.type === 'feed' ? 'Feeding' : 'Poop';
                const volumeStr = log.volume ? log.volume : '';
                
                csvContent += `"${dateStr}","${timeStr}","${typeStr}","${volumeStr}"\n`;
            });

            // Trigger download
            const encodedUri = encodeURI(csvContent);
            const link = document.createElement("a");
            link.setAttribute("href", encodedUri);
            
            // Format filename with current date: baby_logs_YYYY-MM-DD.csv
            const dateSuffix = new Date().toISOString().slice(0, 10);
            link.setAttribute("download", `baby_logs_${dateSuffix}.csv`);
            
            document.body.appendChild(link); // Required for some mobile browsers
            link.click();
            document.body.removeChild(link);
            
            showToast("CSV Exported successfully!");
        }

        // --- UI Updates ---
        function switchTab(tab) {
            currentTab = tab;
            
            // Update Tab UI
            const tabs = ['day', 'week', 'month'];
            tabs.forEach(t => {
                const el = document.getElementById(`tab-${t}`);
                if (t === tab) {
                    el.className = "flex-1 py-2 rounded-xl bg-white shadow text-slate-800 transition-all";
                } else {
                    el.className = "flex-1 py-2 rounded-xl text-slate-500 hover:text-slate-700 transition-all";
                }
            });

            updateDashboard();
        }

        function getStartOf(period) {
            const now = new Date();
            now.setHours(0, 0, 0, 0); // Strictly set to 12:00 AM local time
            
            if (period === 'day') {
                return now.getTime();
            } else if (period === 'week') {
                const day = now.getDay(); // 0 is Sunday
                const diff = now.getDate() - day + (day === 0 ? -6 : 1); // Adjust for Monday start
                const startOfWeek = new Date(now);
                startOfWeek.setDate(diff);
                return startOfWeek.getTime();
            } else if (period === 'month') {
                const startOfMonth = new Date(now);
                startOfMonth.setDate(1);
                return startOfMonth.getTime();
            }
            return 0;
        }

        function updateDashboard() {
            const startTime = getStartOf(currentTab);
            const filteredLogs = logs.filter(log => log.timestamp >= startTime);

            let feedCount = 0;
            let feedVol = 0;
            let poopCount = 0;

            filteredLogs.forEach(log => {
                if (log.type === 'feed') {
                    feedCount++;
                    feedVol += log.volume || 0;
                } else if (log.type === 'poop') {
                    poopCount++;
                }
            });

            document.getElementById('stat-feed-count').innerText = feedCount;
            document.getElementById('stat-feed-vol').innerText = `${feedVol} ml`;
            document.getElementById('stat-poop-count').innerText = poopCount;
        }

        function formatTime(timestamp) {
            const date = new Date(timestamp);
            let hours = date.getHours();
            let minutes = date.getMinutes();
            const ampm = hours >= 12 ? 'PM' : 'AM';
            hours = hours % 12;
            hours = hours ? hours : 12; // the hour '0' should be '12'
            minutes = minutes < 10 ? '0' + minutes : minutes;
            return `${hours}:${minutes} ${ampm}`;
        }

        function formatDate(timestamp) {
            const date = new Date(timestamp);
            const today = new Date();
            if (date.toDateString() === today.toDateString()) return 'Today';
            
            const yesterday = new Date(today);
            yesterday.setDate(yesterday.getDate() - 1);
            if (date.toDateString() === yesterday.toDateString()) return 'Yesterday';

            return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
        }

        function renderHistory() {
            const list = document.getElementById('history-list');
            list.innerHTML = '';

            if (logs.length === 0) {
                list.innerHTML = `<div class="text-center text-slate-400 py-8 text-sm">No entries yet. Time to log!</div>`;
                return;
            }

            // Group by date (simplified for display)
            logs.slice(0, 50).forEach(log => { // Show last 50 for performance
                const isFeed = log.type === 'feed';
                
                const item = document.createElement('div');
                item.className = "bg-white p-4 rounded-2xl shadow-sm border border-slate-100 flex flex-col gap-2";
                
                item.innerHTML = `
                    <div class="flex items-center justify-between">
                        <div class="flex items-center gap-4">
                            <div class="w-12 h-12 rounded-full flex items-center justify-center text-2xl ${isFeed ? 'bg-blue-50 text-blue-500' : 'bg-orange-50 text-orange-500'}">
                                ${isFeed ? '🍼' : '💩'}
                            </div>
                            <div>
                                <p class="font-bold text-slate-700 capitalize">${isFeed ? 'Feeding' : 'Poop'}</p>
                                <p class="text-xs text-slate-400">${formatDate(log.timestamp)} at ${formatTime(log.timestamp)}</p>
                            </div>
                        </div>
                        ${isFeed ? `<div class="text-right">
                            <span class="font-bold text-blue-600">${log.volume}</span>
                            <span class="text-xs text-blue-400">ml</span>
                        </div>` : ''}
                    </div>
                    <div class="flex justify-end gap-3 border-t border-slate-50 pt-2 mt-1">
                        <button onclick="openEditModal('${log.id}')" class="text-xs text-slate-400 font-semibold hover:text-sky-500 transition px-2 py-1">Edit</button>
                        <button onclick="deleteLog('${log.id}', this)" class="text-xs text-slate-400 font-semibold hover:text-red-500 transition px-2 py-1">Delete</button>
                    </div>
                `;
                list.appendChild(item);
            });
        }

        function updateChart() {
            const ctx = document.getElementById('trendsChart').getContext('2d');
            
            // Generate labels and zeroed data for the last 7 days
            const last7Days = [];
            const dataFeed = [];
            const dataPoop = [];
            
            for (let i = 6; i >= 0; i--) {
                const d = new Date();
                d.setHours(0, 0, 0, 0); // Strictly set to 12:00 AM local time
                d.setDate(d.getDate() - i);
                last7Days.push(d);
                dataFeed.push(0);
                dataPoop.push(0);
            }

            // Map the logs to the correct days
            logs.forEach(log => {
                const logDate = new Date(log.timestamp);
                logDate.setHours(0, 0, 0, 0); // Strictly set to 12:00 AM local time
                
                const dayIndex = last7Days.findIndex(d => d.getTime() === logDate.getTime());
                if (dayIndex !== -1) {
                    if (log.type === 'feed') {
                        dataFeed[dayIndex] += (log.volume || 0);
                    } else if (log.type === 'poop') {
                        dataPoop[dayIndex] += 1;
                    }
                }
            });

            // Format dates for the X-axis (e.g., "Mon", "Tue")
            const labels = last7Days.map(d => d.toLocaleDateString('en-US', { weekday: 'short' }));

            // Update existing chart or create a new one
            if (trendsChart) {
                trendsChart.data.labels = labels;
                trendsChart.data.datasets[0].data = dataFeed;
                trendsChart.data.datasets[1].data = dataPoop;
                trendsChart.update();
            } else {
                trendsChart = new Chart(ctx, {
                    type: 'bar',
                    data: {
                        labels: labels,
                        datasets: [
                            {
                                type: 'line',
                                label: 'Feed (ml)',
                                data: dataFeed,
                                borderColor: '#38bdf8', // Tailwind sky-400
                                backgroundColor: '#38bdf8',
                                tension: 0.4,
                                yAxisID: 'y'
                            },
                            {
                                type: 'bar',
                                label: 'Poops',
                                data: dataPoop,
                                backgroundColor: '#fbbf24', // Tailwind amber-400
                                borderRadius: 4,
                                yAxisID: 'y1'
                            }
                        ]
                    },
                    options: {
                        responsive: true,
                        interaction: { mode: 'index', intersect: false },
                        plugins: {
                            legend: {
                                labels: { usePointStyle: true, boxWidth: 8, font: { family: 'Nunito' } }
                            }
                        },
                        scales: {
                            x: { grid: { display: false } },
                            y: { 
                                type: 'linear', 
                                display: true, 
                                position: 'left',
                                title: { display: true, text: 'ml', font: { family: 'Nunito', size: 10 } },
                                border: { display: false }
                            },
                            y1: { 
                                type: 'linear', 
                                display: true, 
                                position: 'right', 
                                grid: { drawOnChartArea: false }, 
                                min: 0, 
                                ticks: { stepSize: 1 },
                                title: { display: true, text: 'count', font: { family: 'Nunito', size: 10 } },
                                border: { display: false }
                            }
                        }
                    }
                });
            }
        }

            poopModal.classList.add('opacity-0');
            poopModalContent.classList.add('translate-y-full');
            
            setTimeout(() => {
                poopModal.classList.add('hidden');
                poopModal.classList.remove('flex');
            }, 300);
        }

        // --- Edit Modal Controls ---
        const editModal = document.getElementById('edit-modal');
        const editModalContent = document.getElementById('edit-modal-content');

        function toggleEditVolume() {
            const type = document.getElementById('edit-type').value;
            const volContainer = document.getElementById('edit-volume-container');
            if (type === 'feed') {
                volContainer.style.display = 'block';
            } else {
                volContainer.style.display = 'none';
            }
        }

        function openEditModal(id) {
            editingLogId = id;
            const log = logs.find(l => l.id === id);
            if(!log) return;
            
            const dateObj = new Date(log.timestamp);
            
            // Format for HTML date/time inputs
            const year = dateObj.getFullYear();
            const month = String(dateObj.getMonth() + 1).padStart(2, '0');
            const day = String(dateObj.getDate()).padStart(2, '0');
            document.getElementById('edit-date').value = `${year}-${month}-${day}`;
            
            const hours = String(dateObj.getHours()).padStart(2, '0');
            const minutes = String(dateObj.getMinutes()).padStart(2, '0');
            document.getElementById('edit-time').value = `${hours}:${minutes}`;

            document.getElementById('edit-type').value = log.type;
            document.getElementById('edit-volume').value = log.volume || '';
            
            toggleEditVolume();

            // Open Animation
            editModal.classList.remove('hidden');
            editModal.classList.add('flex');
            setTimeout(() => {
                editModal.classList.remove('opacity-0');
                editModal.classList.add('opacity-100');
                editModalContent.classList.remove('translate-y-full');
            }, 10);
        }

        function closeEditModal() {
            editModal.classList.remove('opacity-100');
            editModal.classList.add('opacity-0');
            editModalContent.classList.add('translate-y-full');
            
            setTimeout(() => {
                editModal.classList.add('hidden');
                editModal.classList.remove('flex');
                editingLogId = null;
            }, 300);
        }

        function saveEditLog() {
            if(!editingLogId) return;
            
            const dateVal = document.getElementById('edit-date').value;
            const timeVal = document.getElementById('edit-time').value;
            const typeVal = document.getElementById('edit-type').value;
            let volVal = parseInt(document.getElementById('edit-volume').value);

            if(!dateVal || !timeVal) {
                showToast('Please enter a valid date and time');
                return;
            }

            const newTimestamp = new Date(`${dateVal}T${timeVal}`).getTime();

            const logIndex = logs.findIndex(l => l.id === editingLogId);
            if(logIndex !== -1) {
                logs[logIndex].timestamp = newTimestamp;
                logs[logIndex].type = typeVal;
                if(typeVal === 'feed') {
                    logs[logIndex].volume = isNaN(volVal) ? 0 : volVal;
                } else {
                    delete logs[logIndex].volume;
                }
            }
            
            // Re-sort by newest first in case time was changed
            logs.sort((a, b) => b.timestamp - a.timestamp);
            
            saveLogs();
            closeEditModal();
            showToast('Log updated!');
        }

        function deleteLog(id, btnElement) {
            if (btnElement.innerText === "Sure?") {
                logs = logs.filter(l => l.id !== id);
                saveLogs();
                showToast("Log deleted!");
            } else {
                const originalText = btnElement.innerText;
                btnElement.innerText = "Sure?";
                btnElement.classList.add('text-red-500');
                setTimeout(() => {
                    if(document.body.contains(btnElement)) {
                        btnElement.innerText = originalText;
                        btnElement.classList.remove('text-red-500');
                    }
                }, 3000);
            }
        }

        function showToast(message) {
            const toast = document.createElement('div');
            toast.className = "fixed top-8 left-1/2 -translate-x-1/2 bg-slate-800 text-white px-6 py-3 rounded-full shadow-2xl z-50 text-sm font-bold transition-all duration-300 transform translate-y-[-20px] opacity-0";
            toast.innerText = message;
            document.body.appendChild(toast);
            
            setTimeout(() => {
                toast.classList.remove('translate-y-[-20px]', 'opacity-0');
            }, 10);

            setTimeout(() => {
                toast.classList.add('translate-y-[-20px]', 'opacity-0');
                setTimeout(() => toast.remove(), 300);
            }, 2500);
        }

            updateDashboard();
            renderHistory();
            updateChart();
        }
